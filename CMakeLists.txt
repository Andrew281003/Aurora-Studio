cmake_minimum_required(VERSION 3.21)
project(AuroraStudio VERSION 0.1.0 LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Quick QuickControls2 Widgets Test)
qt_add_executable(aurora-studio src/main.cpp src/theme.cpp src/theme.h src/theme_application.cpp src/theme_application.h)
qt_add_resources(aurora-studio "ui" PREFIX "/" FILES qml/Main.qml)
file(GLOB AURORA_THEME_ASSETS CONFIGURE_DEPENDS themes/*.json assets/*.svg assets/*.jpg)
qt_add_resources(aurora-studio "theme-data" PREFIX "/" BASE "${CMAKE_CURRENT_SOURCE_DIR}" FILES ${AURORA_THEME_ASSETS})
target_link_libraries(aurora-studio PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Widgets)
install(TARGETS aurora-studio RUNTIME DESTINATION bin)
configure_file(packaging/aurora-studio.desktop "${CMAKE_CURRENT_BINARY_DIR}/aurora-studio.desktop.in" COPYONLY)
# The portal resolves the desktop ID through its Exec entry, so use the selected install prefix.
install(CODE [[
    file(READ "${CMAKE_CURRENT_LIST_DIR}/aurora-studio.desktop.in" _aurora_desktop_entry)
    string(REPLACE "Exec=aurora-studio" "Exec=${CMAKE_INSTALL_PREFIX}/bin/aurora-studio" _aurora_desktop_entry "${_aurora_desktop_entry}")
    file(MAKE_DIRECTORY "${CMAKE_INSTALL_PREFIX}/share/applications")
    file(WRITE "${CMAKE_INSTALL_PREFIX}/share/applications/aurora-studio.desktop" "${_aurora_desktop_entry}")
]])
add_test(NAME bundled-theme-validation COMMAND ${CMAKE_COMMAND} -E env python3 ${CMAKE_CURRENT_SOURCE_DIR}/tests/validate_themes.py ${CMAKE_CURRENT_SOURCE_DIR})
qt_add_executable(theme-repository-tests tests/theme_repository_test.cpp src/theme.cpp src/theme.h src/theme_application.cpp src/theme_application.h)
qt_add_resources(theme-repository-tests "theme-data" PREFIX "/" BASE "${CMAKE_CURRENT_SOURCE_DIR}" FILES ${AURORA_THEME_ASSETS})
target_link_libraries(theme-repository-tests PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Test)
add_test(NAME theme-repository COMMAND ${CMAKE_COMMAND} -E env QT_QPA_PLATFORM=offscreen $<TARGET_FILE:theme-repository-tests>)
qt_add_executable(qml-wayland-smoke tests/qml_wayland_smoke.cpp src/theme.cpp src/theme.h src/theme_application.cpp src/theme_application.h)
qt_add_resources(qml-wayland-smoke "ui" PREFIX "/" FILES qml/Main.qml)
qt_add_resources(qml-wayland-smoke "theme-data" PREFIX "/" BASE "${CMAKE_CURRENT_SOURCE_DIR}" FILES ${AURORA_THEME_ASSETS})
target_link_libraries(qml-wayland-smoke PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Widgets Qt6::Test)
